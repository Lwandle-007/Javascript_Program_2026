(1) Adding a new Task

START
WHEN user clicks "Add Task" button
    Display the new task form

    Ask user to enter a new task
    Store input as newTask
    // newTask declared with LET because its value changes every time a new task is typed 

    IF user clicks "Add" button THEN
        Save newTask to Firebase
        
        Set createdAt to today's date
        // createdAt declared with CONST - set once, never changes after creation
        Calculate dueDate = createdAt + 7 (days)
        // dueDate declared with CONST - arithmetic operator (+) adds 7 days once, then 
        stays fixed
        Set completed status to false
        // completed declared with LET - its value can later flip to true

        Display the new task on the page (with due date)
        Hide the form

    ELSE IF user clicks "Cancel" button THEN
        Clear the form
        Hide the form
END

(1.1) Filtering Form all Tasks to outstanding/completed 

START

WHEN tasks are loaded from Firebase
    Store all tasks in a list called allTasks
    CALL filterTasksByStatus(allTasks)
    Display returned outstanding list in Outstanding section
    Display returned completed list in Completed section
END

FUNCTION filterTasksByStatus(tasks)
    // PARAMETER: tasks — this function works on ANY task list passed to it,
    // not just allTasks, so it is genuinely reusable
 
    DECLARE const outstandingTasks = tasks.filter(task => task.completed === false)
    DECLARE const completedTasks = tasks.filter(task => task.completed === true)
    // task => task.completed === false is an ARROW FUNCTION passed as a callback
 
    RETURN { outstanding: outstandingTasks, completed: completedTasks }
END FUNCTION


WHEN user clicks a task's circle/checkbox
    Get the clicked task's ID
     CALL toggleTaskStatus(taskId, isCompleted)
END

FUNCTION toggleTaskStatus(taskId, isCompleted)
    // PARAMETERS: taskId and isCompleted — no global variables used inside this 
function
 
    DECLARE let newStatus = isCompleted === false   (comparison operator: ===)
 
    Update task in Firebase using taskId, set completed = newStatus
 
    RETURN newStatus
END FUNCTION
END

(1.3) Summarizing outstanding total and completed total

START

FUNCTION calculateTaskCounts(tasks)
    // PARAMETER: tasks — reusable, does not rely on a global allTasks variable
    DECLARE let outstandingTotal = 0   (local to this function only)
    DECLARE let completedTotal = 0     (local to this function only)

    FOR EACH task IN tasks

        IF task.completed is false THEN
            outstandingTotal = outstandingTotal + 1   (arithmetic operator: +)
        ELSE IF task.completed is true THEN
            completedTotal = completedTotal + 1        (arithmetic operator: +)
        END IF
 
    END FOR
    RETURN { outstanding: outstandingTotal, completed: completedTotal }
END FUNCTION

WHEN allTasks changes (task added, removed, or toggled)
    DECLARE const counts = calculateTaskCounts(allTasks)
 
    CONVERT counts.outstanding (number) to a string
    CONVERT counts.completed (number) to a string
    Display as: counts.outstanding + " outstanding · " + counts.completed + " completed"

END

(3.1) Load and Display Challenge

START
Create list called challenges
    Each challenge has:
        questionTitle
        instructions
        codeSnippet
        options (4 choices)
        correctAnswer

DECLARE let currentChallengeIndex = 0   (LET - increases as user moves through 
challenges)
DECLARE let correctCount = 0            (LET - increases as user answers correctly)
DECLARE let incorrectCount = 0          (LET - increases as user answers incorrectly)

FUNCTION displayChallenge
    Get challenge at currentChallengeIndex
    Show questionTitle
    Show instructions
    Show codeSnippet
    Show the 4 options as buttons
    Show progress: "Challenge (currentChallengeIndex + 1) of 5"
    Show correctCount and incorrectCount at top (/ 0 x 0)
END FUNCTION
END

(3.2) Selecting an Option

START
WHEN user clicks an option button
    Store clicked option as selectedAnswer
    Highlight selectedAnswer (e.g orange background)
    Enable "Check Answer" button
END
END

(3.3) Checking the answer 

START
WHEN user clicks "Check Answer" button
    Get current challenge's correctAnswer

    IF selectedAnswer equals correctAnswer THEN
        // comparison operator: equals
        correctCount = correctCount + 1   (arithmetic operator: +)
        Highlight selectedAnswer green
        Show message "Correct! [short explanation]"
        Change button to "Next Challenge"

    ELSE
        incorrectCount = incorrectCount + 1
        Highlight selectedAnswer red
        Show message "Not quite. [short explanation]"
        Change button to "Next Challenge"
    END IF

    Update / and x counters at top
END
END

(3.4) Moving to Next Challenge

START

WHEN user clicks "Next Challenge"
    Add 1 to currentChallengeIndex

    IF currentChallengeIndex is less than 5 THEN
        Clear selectedAnswer
        CALL displayChallenge
    ELSE
        CALL showResultsScreen
    END IF
END

END

(3.5) Final Results screen

START
FUNCTION calculateScorePercent(correctCount, totalQuestions)
    // PARAMETERS: correctCount and totalQuestions — reusable for any quiz length,
    // not hard-coded to "5" or tied to a global variable
 
    DECLARE let scorePercent = (correctCount / totalQuestions) * 100
    // arithmetic operators: division (/) then multiplication (*)
 
    RETURN scorePercent
END FUNCTION

FUNCTION showResultsScreen(correctCount, incorrectCount, totalQuestions)
    DECLARE const scorePercent = calculateScorePercent(correctCount, totalQuestions)

    Hide challenge card
    Show results icon (books)

    IF scorePercent is greater than or equal to 80 THEN
        // comparison operator: >=
        Show message "Great job!"
    ELSE
        Show message "Keep practising!"
    END IF

    CONVERT correctCount (number) to a string
    CONVERT scorePercent (number) to a string
    Show "You scored " + correctCount + " out of " + totalQuestions + " — " + scorePercent + "%"
    Show correctCount labeled "Correct"
    Show incorrectCount labeled "Incorrect"
    Show "Try Again" button
END FUNCTION
END

    Set currentChallengeIndex to 0
    Set correctCount to 0
    Set incorrectCount to 0
    CALL displayChallenge
END

(4) Dark Mode Toggle(Cookies,dialogs,Redirect_and_print)

START

WHEN page loads
    CALL loadThemePreference()

FUNCTION loadThemePreference()
    // Checks if the user already has a saved preference, so the site
    // remembers their choice next time they visit
    DECLARE let savedTheme = GET "theme" from Firebase (or browser storage)

    IF savedTheme equals "dark" THEN
        // comparison operator: equals
        CALL applyDarkMode()
    ELSE
        CALL applyLightMode()
    END IF

    RETURN savedTheme
END FUNCTION

WHEN user clicks "Dark Mode" toggle
    DECLARE let currentTheme = GET current theme
    DECLARE let newTheme = currentTheme === "dark" ? "light" : "dark"
    // comparison operator (===) combined with a short IF/ELSE choice

    CALL setTheme(newTheme)

FUNCTION setTheme(theme)
    // PARAMETER: theme — reusable, works whether switching to "dark" or "light"
    IF theme equals "dark" THEN
        CALL applyDarkMode()
    ELSE
        CALL applyLightMode()
    END IF

    SAVE "theme" = theme to Firebase (or browser storage)
    // this is what makes the preference persist across visits

    RETURN theme
END FUNCTION

FUNCTION applyDarkMode()
    Change background colour to dark
    Change text colour to light
    Update toggle switch to "on" position
END FUNCTION

FUNCTION applyLightMode()
    Change background colour to light
    Change text colour to dark
    Update toggle switch to "off" position
END FUNCTION

END
